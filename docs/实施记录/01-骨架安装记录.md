## 4. 机器校验（AI 机器验证，可重放）

### 4.1 校验脚本全文（`%TEMP%\dsh-tp-01\validate-preset.cjs`）

```js
// 任务 01 机器校验：preset 组合清单形状 / 行清单 / isolate / 包解析 / 元数据
// 依赖：<DSH_INSTALL>\node_modules\yaml（官方安装目录只读，仅 require，不写入）
const { readFileSync, existsSync } = require('node:fs')
const { join } = require('node:path')
const YAML = require('<DSH_INSTALL>/node_modules/yaml')

const presetDir = '<REPO>/presets/task-planning'
const userDir = '%USERPROFILE%/.dsh/.agent-presets/task-planning'
const base = '<DSH_INSTALL>/node_modules'
const lines = []
let failed = 0
function check(ok, label, detail) {
  lines.push((ok ? 'PASS' : 'FAIL') + ' | ' + label + (detail ? ' | ' + String(detail) : ''))
  if (!ok) failed++
}

// 1. 两个文件在两处都存在（发现阶段只认 agent.cordis.yml，preset.yml 是可选元数据）
for (const [label, dir] of [['仓库内', presetDir], ['用户层', userDir]]) {
  check(existsSync(dir + '/agent.cordis.yml'), label + ' agent.cordis.yml 存在')
  check(existsSync(dir + '/preset.yml'), label + ' preset.yml 存在')
}

const compPath = presetDir + '/agent.cordis.yml'
const text = readFileSync(compPath, 'utf8')

// 2. complete 只允许出现在注释里（硬性要求 2）
const completeLines = text.split('\n').filter(l => /complete/.test(l))
check(completeLines.length > 0 && completeLines.every(l => l.trim().startsWith('#')),
  'complete 仅出现在注释行（无 complete: true）', completeLines.length + ' 处命中，全部为注释')

// 3. 顶层必须是数组（entryListProblem 第 1 条）
let rows = null
try {
  rows = YAML.parse(text)
  check(Array.isArray(rows), '顶层是数组', Array.isArray(rows) ? ('rows=' + rows.length) : ('typeof=' + typeof rows))
} catch (e) { check(false, 'YAML 可解析', e.message) }

// 4. 每一行必须是带非空 name 的 map；group 行的 config 必须是子行列表（entryListProblem 全文）
function rowProblem(list, prefix) {
  if (!Array.isArray(list)) return prefix + ' 不是行列表'
  for (let i = 0; i < list.length; i++) {
    const r = list[i]
    const at = prefix + '第 ' + (i + 1) + ' 行'
    if (r === null || typeof r !== 'object' || Array.isArray(r)) return at + ' 不是 map'
    if (typeof r.name !== 'string' || r.name.trim() === '') return at + ' 缺非空 name'
    if (typeof r.id !== 'string' || r.id.trim() === '') return at + ' 缺 id'
    if (r.group === true) { const p = rowProblem(r.config, at + ' 的 group config '); if (p) return p }
  }
  return null
}
const rp = rowProblem(rows, '')
check(rp === null, '每行都是带非空 name/id 的 map；group 行 config 是子行列表', rp || '无问题')
check(/^[a-z0-9][a-z0-9-]*$/.test('task-planning'), 'preset id 匹配 ^[a-z0-9][a-z0-9-]*$', 'task-planning')

// 5. 压平行清单，逐行列出
const flat = []
;(function walk(list, path) {
  for (const r of (list || [])) {
    flat.push({ path: path + '/' + r.id, name: r.name, group: r.group === true, isolate: r.isolate, config: r.config })
    if (r.group === true) walk(r.config, path + '/' + r.id)
  }
})(rows, '')
lines.push('--- 组合行清单（压平，含 group 子行） ---')
for (const r of flat) lines.push('  ' + r.path.padEnd(28) + ' ->  ' + r.name + (r.group ? '   [group]' : ''))

// 6. 任务 01 要求的 7 行必须在位
const expected = ['@deepseek-ai/dsh-persona', '@deepseek-ai/dsh-tool-ask-user', '@deepseek-ai/dsh-tool-fs',
  '@deepseek-ai/dsh-tool-fs-search', '@deepseek-ai/dsh-compaction-basic', '@deepseek-ai/dsh-command-compact',
  '@deepseek-ai/dsh-compaction-tool-result-pruner']
for (const n of expected) check(flat.some(r => r.name === n), '已挂载 ' + n)

// 7. 压缩组与 isolate
const comp = (rows || []).find(r => r.id === 'compaction')
check(!!comp && comp.group === true, 'compaction 是 cordis:group 行')
check(!!comp && !!comp.isolate && comp.isolate.compaction === true && comp.isolate.toolResultPruner === true,
  'compaction 带 isolate {compaction:true, toolResultPruner:true}', JSON.stringify(comp && comp.isolate))
const children = ((comp && comp.config) || []).map(r => r.name)
check(children.length === 3, '压缩组恰好 3 个子行', children.join(' , '))
const strayIsolate = flat.filter(r => r.isolate && r.path !== '/compaction')
check(strayIsolate.length === 0, '除压缩组外没有任何自创 isolate', strayIsolate.map(r => r.path).join(',') || '无')

// 8. 不给 persona 设 complete 键
const persona = (rows || []).find(r => r.id === 'persona')
const personaKeys = Object.keys((persona && persona.config) || {}).sort()
check(!!persona && !('complete' in (persona.config || {})), 'persona 没有 complete 键', 'config 键 = ' + personaKeys.join(','))
check(!!persona && persona.config.suffix === 'Your working directory is {{cwd}}.', 'persona suffix 带 {{cwd}}', JSON.stringify(persona && persona.config.suffix))

// 9. 包名可解析（模拟发现阶段的健康检查：从 harness base 找 package.json）
for (const n of [...new Set(flat.map(r => r.name))].filter(n => !n.startsWith('cordis:'))) {
  check(existsSync(join(base, n, 'package.json')), 'harness base 可解析包 ' + n)
}

// 10. preset.yml 元数据（只有 name / description / order 三个键）
const meta = YAML.parse(readFileSync(presetDir + '/preset.yml', 'utf8'))
check(typeof meta.name === 'string' && meta.name.trim() !== '', 'preset.yml name 非空字符串', JSON.stringify(meta.name))
check(typeof meta.description === 'string' && meta.description.trim() !== '', 'preset.yml description 非空字符串', JSON.stringify(meta.description))
check(typeof meta.order === 'number' && Number.isFinite(meta.order), 'preset.yml order 是有限数字', JSON.stringify(meta.order))
const extra = Object.keys(meta).filter(k => !['name', 'description', 'order'].includes(k))
check(extra.length === 0, 'preset.yml 只有 name/description/order 三个键', extra.join(',') || '无多余键')

// 11. 用户层副本与仓库内逐字节一致
check(readFileSync(compPath, 'utf8') === readFileSync(userDir + '/agent.cordis.yml', 'utf8'), '用户层 agent.cordis.yml 与仓库内字节一致')
check(readFileSync(presetDir + '/preset.yml', 'utf8') === readFileSync(userDir + '/preset.yml', 'utf8'), '用户层 preset.yml 与仓库内字节一致')

lines.push('')
lines.push(failed === 0 ? '=== 合计：全部 PASS（FAIL 数 0） ===' : ('=== 合计：' + failed + ' 项 FAIL ==='))
console.log(lines.join('\n'))
process.exitCode = failed === 0 ? 0 : 1
```
